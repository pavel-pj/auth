        converter "auth/internal/converter/user"
        desc "auth/pkg/user_v1"
        "context"
)

func (s *Implementation) Index(ctx context.Context, req *desc.IndexRequest) (*desc.IndexResponse, error) {

        limit := req.GetLimit()
        offset := req.GetOffset()
        data, err := s.serv.Index(ctx, limit, offset)
        if err != nil {
                return &desc.IndexResponse{}, err
        }

        var users []*desc.User

        for _, u := range data {
                users = append(users, converter.ToProtoUserFromService(u))
        }

        return &desc.IndexResponse{Users: users}, nil
}

var (
        ErrorPasswwordConfirmation = errors.New("passwords and password confirmation are not equal")
)

type Implementation struct {
        desc.UnimplementedUserV1Server
        serv service.UserService
}

func NewUserImplimentation(userService service.UserService) *Implementation {
        return &Implementation{
                serv: userService,
        }
}

package user

import (
        converter "auth/internal/converter/user"
        desc "auth/pkg/user_v1"
        "context"
)

func (s *Implementation) Store(ctx context.Context, req *desc.StoreRequest) (*desc.StoreResponse, error) {

        if req.GetUser().Password != req.GetUser().PasswordConfirm {
                return &desc.StoreResponse{}, ErrorPasswwordConfirmation
        }

        user := converter.ToUserCreateFromProto(req.GetUser())

        result, err := s.serv.Store(ctx, user)
        if err != nil {
                return &desc.StoreResponse{}, err
        }

        return &desc.StoreResponse{Id: result}, nil
}
