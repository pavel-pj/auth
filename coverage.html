                return &desc.StoreResponse{}, ErrorPasswwordConfirmation

        maxOffset    = 10_000

func (s UserService) Index(ctx context.Context, limit, offset uint64) ([]model.User, error) {

        if limit == 0 {
                limit = defaultLimit
        }
        if limit > maxLimit {
                limit = maxLimit
        }

        users, err := s.Repository.Index(ctx, limit, offset)
        if err != nil {
                return []model.User{}, err
        }

        return users, nil

func (s UserService) Store(ctx context.Context, user model.UserCreate) (int64, error) {

        var id int64

        err := s.txManager.ReadCommitted(ctx, func(ctx context.Context) error {

                var errTx error
                id, errTx = s.Repository.Store(ctx, user)
                if errTx != nil {
                        return errTx
                }

                errTx = s.logService.Store(ctx, model.Log{
                        Level:   model.LevelInfo,
                        Message: fmt.Sprintf("user_store.name:%s,email:%s.role:%s", user.Name, user.Email, user.Roles),
                })

                if errTx != nil {
                        return errTx
                }

                return nil

        })

        if err != nil {
                return 0, err
        }

        return id, nil

}

package user

import (
        "auth/internal/model"
        "fmt"

        "context"
)

func (s UserService) Update(ctx context.Context, user model.UserPatch) (model.User, error) {

        var updatedUser model.User

        err := s.txManager.ReadCommitted(ctx, func(ctx context.Context) error {

                var errTx error
                updatedUser, errTx = s.Repository.Update(ctx, user)
                if errTx != nil {
                        return errTx
                }

                errTx = s.logService.Store(ctx, model.Log{
                        Level:   "info",
                        Message: fmt.Sprintf("user_update.%s", createUpdateLog(user)),
                })
                if errTx != nil {
                        return errTx
                }

                return nil
        })

        if err != nil {
                return model.User{}, err
        }

        return updatedUser, nil

}

func createUpdateLog(user model.UserPatch) string {

        var str string
        if user.Name != nil {
                str = "name: " + *user.Name + ";"
        }
        if user.Email != nil {
                str = str + "enail: " + *user.Email
        }
        if user.Roles != nil {
                str = str + "roles: " + *user.Roles
        }
        return str
}
